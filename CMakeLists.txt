cmake_minimum_required(VERSION 3.20)
project(cdet_renyi_entanglement_hpc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MP_PRECISION_DIGITS10 "100" CACHE STRING "Boost multiprecision decimal digits")
option(CDET_USE_LAPACK "Enable double-precision BLAS/LAPACK kernels" OFF)
option(CDET_NATIVE_ARCH "Optimize for the compiling CPU; not portable across CPU models" OFF)
set(CDET_USE_CHEB_SIMD "0" CACHE STRING "Chebyshev SIMD width (0 disables)")
find_package(Threads REQUIRED)
find_path(BOOST_INCLUDE_DIR boost/multiprecision/cpp_dec_float.hpp
    HINTS ENV BOOST_INCLUDEDIR /opt/homebrew/opt/boost/include /opt/homebrew/include)
find_path(EIGEN_INCLUDE_DIR Eigen/Dense
    HINTS ENV EIGEN_INCLUDE_DIR /opt/homebrew/include/eigen3 /usr/local/include/eigen3 /usr/include/eigen3)
if(NOT BOOST_INCLUDE_DIR OR NOT EIGEN_INCLUDE_DIR)
    message(FATAL_ERROR "Boost and Eigen headers are required; set BOOST_INCLUDE_DIR and EIGEN_INCLUDE_DIR")
endif()
add_executable(cdet_entanglement_entropy src/main.cpp)
target_include_directories(cdet_entanglement_entropy PRIVATE include ${BOOST_INCLUDE_DIR} ${EIGEN_INCLUDE_DIR})
target_compile_definitions(cdet_entanglement_entropy PRIVATE MP_PRECISION_DIGITS10=${MP_PRECISION_DIGITS10})
target_link_libraries(cdet_entanglement_entropy PRIVATE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
    target_compile_options(cdet_entanglement_entropy PRIVATE -Wall -Wextra -Wpedantic)
    if(CDET_NATIVE_ARCH)
        target_compile_options(cdet_entanglement_entropy PRIVATE -march=native)
    endif()
endif()
if(CDET_USE_LAPACK)
    find_package(BLAS REQUIRED)
    find_package(LAPACK REQUIRED)
    target_compile_definitions(cdet_entanglement_entropy PRIVATE CDET_USE_LAPACK=1)
    target_link_libraries(cdet_entanglement_entropy PRIVATE ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES})
endif()
if(NOT CDET_USE_CHEB_SIMD STREQUAL "0")
    target_compile_definitions(cdet_entanglement_entropy PRIVATE CDET_USE_CHEB_SIMD=${CDET_USE_CHEB_SIMD})
endif()
